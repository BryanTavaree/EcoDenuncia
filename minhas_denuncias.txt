  <script>
    const reports = Array.from(document.querySelectorAll(".report"));
    const searchInput = document.getElementById("searchInput");
    const statusFilter = document.getElementById("statusFilter");
    const emptyState = document.getElementById("emptyState");

    function updateReports() {
      const search = searchInput.value.trim().toLowerCase();
      const status = statusFilter.value;
      let visible = 0;

      reports.forEach(report => {
        const searchableText = [
          report.dataset.title,
          report.dataset.location,
          report.dataset.protocol,
          report.querySelector(".report-category").textContent
        ].join(" ").toLowerCase();

        const matchesSearch = searchableText.includes(search);
        const matchesStatus =
          status === "todos" || report.dataset.status === status;

        const show = matchesSearch && matchesStatus;
        report.hidden = !show;

        if (show) visible++;
      });

      emptyState.style.display = visible === 0 ? "block" : "none";
    }

    searchInput.addEventListener("input", updateReports);
    statusFilter.addEventListener("change", updateReports);

    document.querySelectorAll(".details-button").forEach(button => {
      button.addEventListener("click", () => {
        const details = button.closest(".report").querySelector(".details-text");
        details.hidden = !details.hidden;
        button.textContent = details.hidden ? "Ver detalhes" : "Ocultar detalhes";
      });
    });

    document.getElementById("totalCount").textContent = reports.length;
    document.getElementById("analysisCount").textContent =
      reports.filter(report => report.dataset.status === "Em análise").length;
    document.getElementById("completedCount").textContent =
      reports.filter(report => report.dataset.status === "Concluída").length;
  </script>
